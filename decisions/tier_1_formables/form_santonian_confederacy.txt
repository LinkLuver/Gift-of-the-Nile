				tag = ST1
			}
			OR = {
				is_ai = no
			}
			NOR = {
				is_tier_1_formable_trigger = yes
				is_tier_2_formable_trigger = yes

		}
		
		highlight = {

				OR = {
					is_in_area = sentonia_maioris_area

				}
			}
		}
		
		allow = {
			custom_tooltip = {
				text = formable_not_santonian_confederacy_exists

						tag = ST1

		}

		}
