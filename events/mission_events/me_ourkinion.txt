me_ourkinion.4 = { # The Matter of Ourkinion
    type = country_event

    title = me_ourkinion_4_title
    desc = me_ourkinion_4_desc
    picture = city_construction

    immediate = {
        p:3515 = { 
        save_scope_as = alalia_scope
        }
    }
    option = {
        name = "me_ourkinion_4.a"
        custom_tooltip = me_ourkinion_4_tooltip
        add_treasury = 200
        p:3510 = {
            set_city_status = settlement
            add_province_modifier = {
                name = ourkinion_deconstructed
                duration = 3650
            }
        }
        p:3510 = {
            ordered_pops_in_province = {
                order_by = pop_hapiness
                max = {
                    value = total_population
                    multiply = 0.7
                    round = floor
                }
                check_range_bounds = no
                move_pop = scope:alalia_scope
            }
        }
    }
    option = {
        name = "me_ourkinion_4.b"
        custom_tooltip = me_ourkinion_4_tooltip
        add_treasury = -150
        add_manpower = -5
        p:3510 = {
            add_province_modifier = {
                name = ourkinion_training
                duration = -1
            }
            create_state_pop = citizen
            create_state_pop = freemen
        }
    }
}

me_ourkinion.6 = { # On the Corsians
    type = country_event

    title = me_ourkinion_6_title
    desc = me_ourkinion_6_desc
    picture = greek_political_discussion

    option = {
        custom_tooltip = me_ourkinion_6_tooltip
        name = "me_ourkinion_6.a"
        add_country_modifier = {
            name = retained_hellenic_ways 
            duration = 7300
        }
        random_country_culture = {
			limit = {
				is_culture = corsian
			}
			add_country_culture_modifier = {
				name = corsian_dissent
				duration = 7300
			}
            set_variable = corsians_shunned
		}
    }
    option = {
        custom_tooltip = me_ourkinion_6_tooltip
        name = "me_ourkinion_6.b"
        add_country_modifier = {
            name = accepted_corsians
            duration = 7300
        }
        random_country_culture = {
            limit = {
                is_culture = corsian
            }
            set_country_culture_right = citizen
            add_integration_progress = 1
        }
        set_variable = corsians_integrated
    }
}
